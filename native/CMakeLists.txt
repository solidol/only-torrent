cmake_minimum_required(VERSION 3.20)
project(ltbridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ltbridge requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Both C++ DLLs share the same CRT. The Delphi boundary exposes POD only.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BOOST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../deps/boost_1_78_0" CACHE PATH "Pinned Boost headers")
set(Boost_NO_SYSTEM_PATHS ON)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL ON)
set(CMAKE_DISABLE_FIND_PACKAGE_LibGcrypt ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Iconv ON)
# Link ONLY an installed upstream SDK; this build never compiles libtorrent.
find_package(LibtorrentRasterbar CONFIG REQUIRED)
add_library(ltbridge SHARED ltbridge.cpp)
target_link_libraries(ltbridge PRIVATE LibtorrentRasterbar::torrent-rasterbar)
target_compile_definitions(ltbridge PRIVATE LTBRIDGE_BUILD _WIN32_WINNT=0x0600 WINVER=0x0600)
target_compile_options(ltbridge PRIVATE /W4 /O2 /utf-8)
set_target_properties(ltbridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/package")
